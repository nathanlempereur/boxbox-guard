# 1. Création du lien symbolique
ln -s /etc/boxbox-guard/boxbox-guard.service /etc/systemd/system/boxbox-guard.service

# 2. Recharger systemd pour qu'il voit le nouveau fichier
systemctl daemon-reload

# 3. Activer le lancement automatique au démarrage et le démarrer maintenant
systemctl enable --now boxbox-guard
